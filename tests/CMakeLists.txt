function(configure_simulator_test test_name)
    if(MSVC)
        target_compile_options(${test_name} PRIVATE /W4 /permissive-)
        if(SIMULATOR_WARNINGS_AS_ERRORS)
            target_compile_options(${test_name} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${test_name} PRIVATE -Wall -Wextra -Wpedantic)
        if(SIMULATOR_WARNINGS_AS_ERRORS)
            target_compile_options(${test_name} PRIVATE -Werror)
        endif()
    endif()
endfunction()

function(add_simulator_v1_test test_name)
    add_executable(${test_name} ${test_name}.cpp)
    target_link_libraries(${test_name} PRIVATE simulator_core simulator_scheduler)
    configure_simulator_test(${test_name})
    add_test(NAME ${test_name} COMMAND ${test_name})
endfunction()
